import time import random def king_kai_training(): print("Welcome to King Kai Training! 🐝") # Get user input duration = input("How many minutes do you want to train? (1-5): ") intensity = input("Choose intensity from 1 (easy) to 10 (hard): ") if not (duration.isdigit() and intensity.isdigit()): print("Please enter numbers only. Try again.") return duration = int(duration) intensity = int(intensity) if duration < 1 or duration > 5 or intensity < 1 or intensity > 10: print("Please enter a duration between 1-5 and intensity between 1-10. Try again.") return # Realistic exercises inspired by King Kai training exercises = [ ("Bumblebee Catch", "Try to move quickly side to side, dodging imaginary bumblebees for agility."), ("Gregory Dodge", "Move around slowly, practicing quick direction changes to dodge Gregory."), ("King Kai Push-ups", "Do controlled push-ups to build strength and endurance."), ("Energy Focus Breathing", "Practice deep breathing, focusing your energy to calm your mind."), ("Speed Jumps", "Jump in place quickly and repeatedly to build explosive power."), ("Balance Pose", "Stand on one foot, switch feet, to improve balance and focus."), ] # Select exercises based on intensity if intensity <= 3: chosen = [exercises[1], exercises[3], exercises[5]] # easier ones elif intensity <= 7: chosen = [exercises[0], exercises[2], exercises[4]] # moderate else: chosen = exercises # all, for high intensity print(f"\nYour King Kai Training session for {duration} minute(s) at intensity {intensity} begins!") print("Follow the current exercise description, then you'll switch every interval.\n") intervals = len(chosen) base_interval = (duration * 60) // intervals # Higher intensity = tighter, punchier work intervals (floor at 5s) interval_time = max(5, base_interval - intensity) # Higher intensity = shorter rest between exercises (floor at 2s) rest_time = max(2, 10 - intensity) for i, (name, desc) in enumerate(chosen, 1): print(f"Exercise {i}: {name}") print(f" {desc}") print(f"Start now! Training for {interval_time} seconds...") time.sleep(interval_time) if i < len(chosen): print(f"Rest for {rest_time} seconds...\n") time.sleep(rest_time) print("\nGreat job! Your King Kai training is complete. Keep practicing to build your energy and skill! 💥") king_kai_training()
